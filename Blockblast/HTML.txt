<script>

const SIZE = 8;

let board = [];

let score = 0;

let pieces = [];

let selectedPiece = null;


/* COLORS */

const colors = [
    "#ff4757",
    "#ffa502",
    "#2ed573",
    "#1e90ff",
    "#a55eea",
    "#ff6b81",
    "#00d2d3"
];


/* BLOCK SHAPES */

const shapes = [

    // Single block
    [
        [0,0]
    ],

    // 2 horizontal
    [
        [0,0],
        [0,1]
    ],

    // 3 horizontal
    [
        [0,0],
        [0,1],
        [0,2]
    ],

    // 4 horizontal
    [
        [0,0],
        [0,1],
        [0,2],
        [0,3]
    ],

    // 5 horizontal
    [
        [0,0],
        [0,1],
        [0,2],
        [0,3],
        [0,4]
    ],

    // Vertical 3
    [
        [0,0],
        [1,0],
        [2,0]
    ],

    // Square
    [
        [0,0],
        [0,1],
        [1,0],
        [1,1]
    ],

    // L
    [
        [0,0],
        [1,0],
        [2,0],
        [2,1]
    ],

    // Reverse L
    [
        [0,1],
        [1,1],
        [2,1],
        [2,0]
    ],

    // T
    [
        [0,0],
        [0,1],
        [0,2],
        [1,1]
    ],

    // Z
    [
        [0,0],
        [0,1],
        [1,1],
        [1,2]
    ],

    // Plus
    [
        [0,1],
        [1,0],
        [1,1],
        [1,2],
        [2,1]
    ],

    // 3x3 square
    [
        [0,0],
        [0,1],
        [0,2],

        [1,0],
        [1,1],
        [1,2],

        [2,0],
        [2,1],
        [2,2]
    ]
];


/* CREATE EMPTY BOARD */

function createBoard() {

    board = [];

    for (let row = 0; row < SIZE; row++) {

        board[row] = [];

        for (let col = 0; col < SIZE; col++) {

            board[row][col] = null;

        }
    }
}


/* DRAW BOARD */

function drawBoard() {

    const boardElement =
        document.getElementById("board");

    boardElement.innerHTML = "";


    for (let row = 0; row < SIZE; row++) {

        for (let col = 0; col < SIZE; col++) {

            const cell =
                document.createElement("div");

            cell.className = "cell";

            cell.dataset.row = row;
            cell.dataset.col = col;


            if (board[row][col]) {

                cell.classList.add("filled");

                cell.style.background =
                    board[row][col];

            }


            cell.addEventListener(
                "click",
                () => {

                    if (selectedPiece !== null) {

                        placePiece(
                            selectedPiece,
                            row,
                            col
                        );

                    }

                }
            );


            boardElement.appendChild(cell);

        }
    }
}


/* CREATE RANDOM PIECE */

function randomPiece() {

    const shape =
        shapes[
            Math.floor(
                Math.random() * shapes.length
            )
        ];


    const color =
        colors[
            Math.floor(
                Math.random() * colors.length
            )
        ];


    return {

        shape: shape,

        color: color

    };
}


/* GENERATE THREE PIECES */

function generatePieces() {

    pieces = [

        randomPiece(),

        randomPiece(),

        randomPiece()

    ];


    drawPieces();
}


/* DRAW PIECES */

function drawPieces() {

    const container =
        document.getElementById("pieces");

    container.innerHTML = "";


    pieces.forEach((piece, index) => {

        const element =
            document.createElement("div");

        element.className = "piece";


        const maxRow =
            Math.max(
                ...piece.shape.map(
                    block => block[0]
                )
            );


        const maxCol =
            Math.max(
                ...piece.shape.map(
                    block => block[1]
                )
            );


        element.style.gridTemplateRows =
            `repeat(${maxRow + 1}, 25px)`;


        element.style.gridTemplateColumns =
            `repeat(${maxCol + 1}, 25px)`;


        for (
            let row = 0;
            row <= maxRow;
            row++
        ) {

            for (
                let col = 0;
                col <= maxCol;
                col++
            ) {

                const cell =
                    document.createElement("div");


                const exists =
                    piece.shape.some(
                        block =>
                            block[0] === row &&
                            block[1] === col
                    );


                if (exists) {

                    cell.className =
                        "piece-cell";

                    cell.style.background =
                        piece.color;

                } else {

                    cell.className =
                        "empty-piece-cell";

                }


                element.appendChild(cell);

            }
        }


        element.addEventListener(
            "click",
            () => {

                selectedPiece = index;

            }
        );


        container.appendChild(element);

    });
}


/* CHECK IF PIECE CAN FIT */

function canPlace(
    piece,
    row,
    col
) {

    for (const [r, c] of piece.shape) {

        const newRow =
            row + r;

        const newCol =
            col + c;


        // Outside board

        if (
            newRow < 0 ||
            newRow >= SIZE ||
            newCol < 0 ||
            newCol >= SIZE
        ) {

            return false;

        }


        // Cell already occupied

        if (
            board[newRow][newCol] !== null
        ) {

            return false;

        }

    }


    return true;
}


/* PLACE PIECE */

function placePiece(
    pieceIndex,
    row,
    col
) {

    const piece =
        pieces[pieceIndex];


    if (!piece) {
        return;
    }


    if (
        !canPlace(
            piece,
            row,
            col
        )
    ) {

        return;

    }


    /* Put blocks on board */

    for (
        const [r, c]
        of piece.shape
    ) {

        board[row + r][col + c] =
            piece.color;

    }


    /* Add score */

    score +=
        piece.shape.length * 10;


    /* Remove used piece */

    pieces.splice(
        pieceIndex,
        1
    );


    selectedPiece = null;


    /* Clear lines */

    clearLines();


    /* If all three pieces used */

    if (pieces.length === 0) {

        generatePieces();

    } else {

        drawPieces();

    }


    drawBoard();

    updateScore();

    checkGameOver();

}


/* CLEAR FULL ROWS AND COLUMNS */

function clearLines() {

    let cleared = 0;


    /* ROWS */

    for (
        let row = SIZE - 1;
        row >= 0;
        row--
    ) {

        const full =
            board[row].every(
                cell => cell !== null
            );


        if (full) {

            board.splice(
                row,
                1
            );


            board.unshift(
                new Array(SIZE).fill(null)
            );


            cleared++;

            row++;

        }

    }


    /* COLUMNS */

    for (
        let col = 0;
        col < SIZE;
        col++
    ) {

        let full = true;


        for (
            let row = 0;
            row < SIZE;
            row++
        ) {

            if (
                board[row][col] === null
            ) {

                full = false;

                break;

            }

        }


        if (full) {

            for (
                let row = 0;
                row < SIZE;
                row++
            ) {

                board[row][col] = null;

            }


            cleared++;

        }

    }


    /* BONUS */

    if (cleared > 0) {

        score +=
            cleared * 100;

    }

}


/* CHECK GAME OVER */

function checkGameOver() {

    for (
        let i = 0;
        i < pieces.length;
        i++
    ) {

        const piece =
            pieces[i];


        for (
            let row = 0;
            row < SIZE;
            row++
        ) {

            for (
                let col = 0;
                col < SIZE;
                col++
            ) {

                if (
                    canPlace(
                        piece,
                        row,
                        col
                    )
                ) {

                    return;

                }

            }

        }

    }


    /* No possible moves */

    document.getElementById(
        "finalScore"
    ).textContent = score;


    document.getElementById(
        "gameOver"
    ).style.display = "flex";

}


/* UPDATE SCORE */

function updateScore() {

    document.getElementById(
        "score"
    ).textContent = score;

}


/* RESTART */

function restartGame() {

    score = 0;

    selectedPiece = null;


    document.getElementById(
        "gameOver"
    ).style.display = "none";


    createBoard();

    generatePieces();

    drawBoard();

    updateScore();

}


/* START GAME */

restartGame();

</script>
